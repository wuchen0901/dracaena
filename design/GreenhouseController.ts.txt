import { _decorator, AudioClip, AudioSource, Camera, Color, Component, EventKeyboard, Graphics, Input, input, KeyCode, Label, Mask, Node, resources, UITransform, Vec3 } from 'cc';
const { ccclass } = _decorator;
type Platform = { x: number; w: number; y: number };

/** One original greenhouse journey, drawn in layers in a 1280 × 720 canvas. */
@ccclass('PuzzleController')
export class PuzzleController extends Component {
  private g!: Graphics;
  private title!: Label;
  private objective!: Label;
  private prompt!: Label;
  private controls!: Label;
  private ending!: Label;
  private keys = new Set<KeyCode>();
  private x = 150; private y = -210; private vx = 0; private vy = 0;
  private facing = 1; private grounded = true; private coyote = 0; private jumpBuffer = 0;
  private camera = 640; private time = 0; private stepTimer = 0;
  private carrying = false; private powered = false; private operating = false;
  private angle = -12; private charge = 0; private gateOpen = false; private won = false;
  private seedX = 480; private seedY = -180;
  private readonly mirrorX = 1370; private readonly mirrorY = -112;
  private readonly receiverX = 1720; private readonly receiverY = 105;
  private readonly gateX = 2160;
  private readonly platforms: Platform[] = [
    { x: -100, w: 950, y: -210 }, { x: 1050, w: 1500, y: -210 },
    { x: 790, w: 145, y: -145 }, { x: 960, w: 145, y: -85 },
  ];
  private audio!: AudioSource;
  private wind: AudioClip | null = null; private step: AudioClip | null = null; private chime: AudioClip | null = null;
  private audioStarted = false;

  onLoad() {
    const mainCamera = this.node.scene.getChildByName('Main Camera')?.getComponent(Camera);
    if (mainCamera) { mainCamera.clearFlags = Camera.ClearFlag.SOLID_COLOR; mainCamera.clearColor = new Color(9, 24, 23, 255); }
    const frame = this.child('Landscape frame');
    frame.addComponent(UITransform).setContentSize(1280, 720);
    frame.addComponent(Mask).type = Mask.Type.GRAPHICS_RECT;
    const art = new Node('Greenhouse landscape'); art.setParent(frame); art.layer = this.node.layer;
    art.addComponent(UITransform).setContentSize(1280, 720);
    this.g = art.addComponent(Graphics);
    this.title = this.label('Chapter', 'THE LAST LIGHT  /  THE GLASS GARDEN', -350, 310, 17, 560);
    this.objective = this.label('Objective', 'Find the living light', 390, 310, 16, 420);
    this.prompt = this.label('Interaction', '', 0, -265, 18, 1000);
    this.controls = this.label('Controls', 'A / D  MOVE     SPACE  JUMP     E  INTERACT     R  RESTART', 0, -320, 14, 1100);
    this.ending = this.label('Ending', '', 0, 60, 38, 1100);
    this.audio = this.node.addComponent(AudioSource);
    resources.load('audio/wind', AudioClip, (err, clip) => { if (!err) this.wind = clip; });
    resources.load('audio/step', AudioClip, (err, clip) => { if (!err) this.step = clip; });
    resources.load('audio/chime', AudioClip, (err, clip) => { if (!err) this.chime = clip; });
    input.on(Input.EventType.KEY_DOWN, this.keyDown, this);
    input.on(Input.EventType.KEY_UP, this.keyUp, this);
    this.draw();
  }
  onDestroy() {
    input.off(Input.EventType.KEY_DOWN, this.keyDown, this);
    input.off(Input.EventType.KEY_UP, this.keyUp, this);
  }
  private child(name: string) { const n = new Node(name); n.setParent(this.node); n.layer = this.node.layer; return n; }
  private label(name: string, text: string, x: number, y: number, size: number, width: number) {
    const n = this.child(name); n.setPosition(x, y, 0); n.addComponent(UITransform).setContentSize(width, 70);
    const l = n.addComponent(Label); l.string = text; l.fontSize = size; l.lineHeight = size + 12;
    l.color = new Color(232, 239, 214); l.horizontalAlign = Label.HorizontalAlign.CENTER;
    return l;
  }
  private keyDown(e: EventKeyboard) {
    const fresh = !this.keys.has(e.keyCode); this.keys.add(e.keyCode);
    if (!this.audioStarted && this.wind) { this.audio.clip = this.wind; this.audio.loop = true; this.audio.volume = 0.14; this.audio.play(); this.audioStarted = true; }
    if (!fresh) return;
    if (e.keyCode === KeyCode.KEY_R) this.reset();
    if ([KeyCode.SPACE, KeyCode.KEY_W, KeyCode.ARROW_UP].includes(e.keyCode)) this.jumpBuffer = 0.16;
    if (e.keyCode === KeyCode.KEY_E) this.interact();
  }
  private keyUp(e: EventKeyboard) { this.keys.delete(e.keyCode); }
  private interact() {
    if (this.won) return;
    if (!this.powered && !this.carrying && Math.abs(this.x - this.seedX) < 85 && Math.abs(this.y - (-210)) < 90) {
      this.carrying = true; if (this.chime) this.audio.playOneShot(this.chime, 0.35); return;
    }
    if (Math.abs(this.x - this.mirrorX) < 100 && this.grounded) {
      if (this.carrying) { this.carrying = false; this.powered = true; if (this.chime) this.audio.playOneShot(this.chime, 0.5); }
      else if (this.powered && !this.gateOpen) this.operating = !this.operating;
    }
  }
  private reset() {
    this.x = 150; this.y = -210; this.vx = this.vy = 0; this.camera = 640; this.grounded = true;
    this.carrying = this.powered = this.operating = this.gateOpen = this.won = false;
    this.angle = -12; this.charge = this.jumpBuffer = this.coyote = 0; this.ending.string = '';
  }
  update(delta: number) {
    const dt = Math.min(delta, 1 / 30); this.time += dt;
    const left = this.keys.has(KeyCode.KEY_A) || this.keys.has(KeyCode.ARROW_LEFT);
    const right = this.keys.has(KeyCode.KEY_D) || this.keys.has(KeyCode.ARROW_RIGHT);
    const direction = Number(right) - Number(left);
    this.jumpBuffer = Math.max(0, this.jumpBuffer - dt);
    this.coyote = this.grounded ? 0.12 : Math.max(0, this.coyote - dt);
    if (!this.won) {
      if (this.operating) { this.vx = 0; this.angle = Math.max(-35, Math.min(70, this.angle + direction * 26 * dt)); }
      else {
        const target = direction * (this.carrying ? 205 : 235);
        this.vx += (target - this.vx) * Math.min(1, dt * (direction ? 12 : 16));
        if (direction) this.facing = direction;
        if (this.jumpBuffer > 0 && this.coyote > 0) { this.vy = 570; this.grounded = false; this.coyote = this.jumpBuffer = 0; }
        this.x = Math.max(35, Math.min(this.gateOpen ? 2390 : this.gateX - 35, this.x + this.vx * dt));
        const previous = this.y; this.vy -= 1450 * dt; this.y += this.vy * dt; this.grounded = false;
        if (this.vy <= 0) for (const p of this.platforms) {
          if (this.x + 12 > p.x && this.x - 12 < p.x + p.w && previous >= p.y - 1 && this.y <= p.y) {
            this.y = p.y; this.vy = 0; this.grounded = true; break;
          }
        }
        if (this.y < -490) { this.x = 680; this.y = -210; this.vx = this.vy = 0; this.grounded = true; }
        if (this.grounded && Math.abs(this.vx) > 70) {
          this.stepTimer -= dt; if (this.stepTimer <= 0) { if (this.step) this.audio.playOneShot(this.step, 0.22); this.stepTimer = 0.32; }
        }
      }
      if (this.powered && !this.gateOpen) {
        const targetAngle = Math.atan2(this.receiverY - this.mirrorY, this.receiverX - this.mirrorX) * 90 / Math.PI;
        const aligned = Math.abs(this.angle - targetAngle) < 3.5;
        this.charge = Math.max(0, Math.min(1, this.charge + (aligned ? dt / 0.8 : -dt * 2)));
        if (this.charge >= 1) { this.gateOpen = true; this.operating = false; this.angle = targetAngle; if (this.chime) this.audio.playOneShot(this.chime, 0.8); }
      }
      if (this.gateOpen && this.x > this.gateX + 100) { this.won = true; this.vx = 0; this.ending.string = 'Somewhere beyond the glass,\nmorning began.'; }
    }
    const cameraTarget = Math.max(640, Math.min(1810, this.x + 180));
    this.camera += (cameraTarget - this.camera) * (1 - Math.exp(-dt * 3));
    this.objective.string = this.won ? 'CHAPTER COMPLETE' : this.gateOpen ? 'Follow the light beyond the gate' : this.powered ? 'Wake the receiver' : this.carrying ? 'Bring the light to the old mirror' : 'Find the living light';
    this.prompt.string = this.won ? 'R  ·  BEGIN AGAIN' : this.operating ? this.charge > 0 ? 'HOLD THE LIGHT STEADY…' : '← / →  AIM THE MIRROR     E  STEP AWAY' : !this.powered && !this.carrying && Math.abs(this.x - this.seedX) < 85 ? 'E  ·  TAKE THE LIGHT' : Math.abs(this.x - this.mirrorX) < 100 ? this.carrying ? 'E  ·  OFFER THE LIGHT' : this.powered && !this.gateOpen ? 'E  ·  TURN THE MIRROR' : '' : '';
    this.draw();
  }
  private sx(x: number, parallax = 1) { return x - this.camera * parallax - (1 - parallax) * 640; }
  private color(hex: string, a = 255) { const v = parseInt(hex, 16); return new Color(v >> 16, (v >> 8) & 255, v & 255, a); }
  private rect(x: number, y: number, w: number, h: number, hex: string, a = 255) { this.g.fillColor = this.color(hex, a); this.g.rect(x, y, w, h); this.g.fill(); }
  private circle(x: number, y: number, r: number, hex: string, a = 255) { this.g.fillColor = this.color(hex, a); this.g.circle(x, y, r); this.g.fill(); }
  private line(x: number, y: number, xx: number, yy: number, hex: string, width = 2, a = 255) { this.g.strokeColor = this.color(hex, a); this.g.lineWidth = width; this.g.moveTo(x, y); this.g.lineTo(xx, yy); this.g.stroke(); }
  private polygon(points: number[][], hex: string, a = 255) { this.g.fillColor = this.color(hex, a); this.g.moveTo(points[0][0], points[0][1]); for (const p of points.slice(1)) this.g.lineTo(p[0], p[1]); this.g.close(); this.g.fill(); }
  private glow(x: number, y: number, r: number, strength = 1) { for (let i = 5; i >= 1; i--) this.circle(x, y, r * i / 5, 'f5df9c', (6 + (5 - i) * 5) * strength); }
  private tree(worldX: number, base: number, height: number, parallax: number, dark: boolean) {
    const x = this.sx(worldX, parallax); const trunk = dark ? '173f3b' : '4b8171';
    this.polygon([[x-25,base],[x+30,base],[x+14,base+height*.7],[x+45,base+height],[x+10,base+height*.91],[x-10,base+height*.56]],trunk);
    for(let i=0;i<18;i++) { const t=i*2.399; const r=35+(i%5)*21; const xx=x+Math.cos(t)*r; const yy=base+height-20+Math.sin(t)*r*.48;
      this.circle(xx,yy,37+(i%3)*15,dark ? ['245f50','306e56','418365'][i%3] : ['6b9f80','86b08b','9dbb91'][i%3]); }
  }
  private plant(x: number, y: number, size: number, bright: boolean) {
    const sway = Math.sin(this.time * 1.4 + x*.02)*4;
    this.line(x,y,x+sway,y+size,bright?'89b56c':'426d51',2);
    for(let i=0;i<3;i++) { const yy=y+size*(.25+i*.22), side=i%2?1:-1;
      this.polygon([[x,yy],[x+side*size*.35,yy+size*.17],[x+side*size*.4,yy+size*.32],[x+sway,yy+size*.14]],bright?'a6ca75':'527d56'); }
  }
  private draw() {
    const g=this.g; g.clear();
    // Sky is painted as overlapping bands, with mist separating the landscape layers.
    this.rect(-640,-360,1280,720,'79afb0');
    for(let i=0;i<14;i++) this.rect(-640,-360+i*52,1280,53,i<5?'b3c8a1':i<9?'9abfb0':'76adb0',28);
    this.circle(350-this.camera*.025,220,92,'d3dfbf',65);
    for(let i=0;i<11;i++) { const x=this.sx(i*250, .08); this.circle(x,180+(i%3)*32,48,'e1e5c8',65); this.circle(x+55,185+(i%3)*32,65,'e1e5c8',55); }
    for(let layer=0;layer<3;layer++) {
      const p=.12+layer*.13; const pts:number[][]=[[-640,-300]];
      for(let i=0;i<=28;i++) {const x=-660+i*50; pts.push([x,-40-layer*54+Math.sin((x+this.camera*p)*.008)*55+Math.sin((x+this.camera*p)*.021)*20]);}
      pts.push([700,-360]); this.polygon(pts,['8eaf96','729b83','507e6a'][layer]);
    }
    for(let i=0;i<10;i++) this.tree(i*350-150,-190,240+(i%3)*40,.4,false);
    // A roof of broken glass and iron ribs encloses the playable garden.
    for(let i=0;i<9;i++) { const x=this.sx(i*330); g.strokeColor=this.color('335d56',150); g.lineWidth=5;
      g.moveTo(x,-210); g.lineTo(x+18,165); g.quadraticCurveTo(x+165,330,x+312,165); g.lineTo(x+330,-210); g.stroke();
      this.line(x+18,165,x+312,165,'476e60',2,120);
      if(i%2===0) this.polygon([[x+32,178],[x+110,240],[x+130,186]],'d2dfc1',24);
    }
    for(let i=0;i<5;i++) this.tree(i*630-140,-220,390+(i%2)*55,.78,true);
    // Traversable rock shelves; collision uses the same platform data.
    for(const p of this.platforms) {const x=this.sx(p.x);
      this.polygon([[x,p.y],[x+p.w,p.y],[x+p.w-24,p.y-180],[x+30,p.y-200]],'243f3b');
      this.rect(x,p.y-8,p.w,9,'718b5a'); this.rect(x,p.y-18,p.w,10,'435d47');
      for(let j=0;j<p.w;j+=95) this.line(x+j,p.y-25,x+j+35,p.y-78,'35514a',3);
    }
    for(let i=0;i<52;i++) {const wx=i*48; if(wx>850&&wx<1050)continue; this.plant(this.sx(wx),-210,18+(i%4)*9,this.gateOpen&&wx>1500);}
    // Dormant seed and mirror pedestal.
    if(!this.carrying&&!this.powered) {const x=this.sx(this.seedX); this.glow(x,this.seedY,52); this.circle(x,this.seedY,7,'fff1b3');}
    const mx=this.sx(this.mirrorX), my=this.mirrorY;
    this.rect(mx-22,-210,44,72,'50685a'); this.rect(mx-40,-150,80,12,'7e8b6b');
    const sourceX=mx-145;
    this.rect(sourceX-18,-210,36,76,'435c51'); this.circle(sourceX,my,14,this.powered?'f9e9ab':'71836b');
    if(this.powered) {
      this.glow(sourceX,my,45); this.line(sourceX,my,mx,my,'f5df9b',4,210);
      const ray=this.angle*Math.PI/90, length=this.gateOpen?Math.hypot(this.receiverX-this.mirrorX,this.receiverY-my):700;
      this.line(mx,my,mx+Math.cos(ray)*length,my+Math.sin(ray)*length,'f9e8ac',12,25);
      this.line(mx,my,mx+Math.cos(ray)*length,my+Math.sin(ray)*length,'fff0b6',3,225);
    }
    const a=this.angle*Math.PI/180, tx=Math.cos(a)*50, ty=Math.sin(a)*50;
    this.line(mx-tx,my-ty,mx+tx,my+ty,'344e49',15); this.line(mx-tx,my-ty,mx+tx,my+ty,'bdd9c5',7); this.circle(mx,my,5,'e6dba3');
    const rx=this.sx(this.receiverX),ry=this.receiverY;
    this.line(rx,-210,rx,ry-25,'3d6157',12); this.circle(rx,ry,27,'45685b');
    g.strokeColor=this.color(this.gateOpen?'f9e4a0':'98b18b'); g.lineWidth=3;g.circle(rx,ry,22);g.stroke();
    if(this.charge>0) {this.glow(rx,ry,60,this.charge);this.circle(rx,ry,12,'fff1ba',255*this.charge);}
    const gx=this.sx(this.gateX);
    if(this.gateOpen) {this.glow(gx,-125,125,.7);this.rect(gx-37,-210,74,150,'d7d9a0',65);}
    g.strokeColor=this.color(this.gateOpen?'d4d5a0':'5d7862');g.lineWidth=12;g.moveTo(gx-50,-210);g.lineTo(gx-50,-85);g.quadraticCurveTo(gx,-25,gx+50,-85);g.lineTo(gx+50,-210);g.stroke();
    if(!this.gateOpen) for(let i=-30;i<=30;i+=15)this.line(gx+i,-210,gx+i,-82,'335349',6);
    // Small readable character, with a scarf and a walk cycle.
    const px=this.sx(this.x), py=this.y, moving=this.grounded&&Math.abs(this.vx)>30;
    const stride=moving?Math.sin(this.time*13)*12:0, bob=moving?Math.abs(Math.sin(this.time*13))*2:Math.sin(this.time*2)*.8;
    this.circle(px,py+3,15,'172e2b',65);
    this.line(px-3,py+26,px-6+stride,py+3,'172f32',7);this.line(px+3,py+26,px+6-stride,py+3,'213a3a',7);
    this.polygon([[px-10,py+53+bob],[px+9,py+53+bob],[px+12,py+25],[px-12,py+25]],'ddd9b9');
    this.line(px-this.facing*6,py+47,px+this.facing*16,py+36,'d1c9a6',5);
    this.circle(px,py+63+bob,10,'cfb798');this.circle(px-3,py+68+bob,10,'203d3b');
    this.polygon([[px+this.facing*6,py+67+bob],[px+this.facing*13,py+62+bob],[px+this.facing*7,py+58+bob]],'cfb798');
    this.polygon([[px-8,py+52+bob],[px-26-this.facing*5,py+45+bob+Math.sin(this.time*5)*3],[px-13,py+44+bob]],'bf7952');
    if(this.carrying){this.glow(px+this.facing*17,py+42,50);this.circle(px+this.facing*17,py+42,6,'fff2b9');}
    // Near foliage and drifting motes sell depth without hiding the playable silhouette.
    for(let i=0;i<18;i++){const x=this.sx(i*180,1.12);this.plant(x,-350,75+(i%4)*20,false);}
    for(let i=0;i<36;i++){const x=((i*173+this.time*(3+i%3))%2800);this.circle(this.sx(x),-150+(i*83)%400+Math.sin(this.time+i)*8,1.5,'e5e8b1',65+Math.sin(this.time+i)*45);}
    this.rect(-640,280,1280,80,'102e2a',90); this.rect(-640,-360,1280,78,'102a26',200);
    if(this.won) this.rect(-640,-360,1280,640,'143e35',150);
  }
}
